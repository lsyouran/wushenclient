# 13 · 指令系统与分发链路

> 范围：`os/command.js`（46 行）+ `os/char/character.js:136-222` + `world/cmd/`（123 文件）

---

## 13.1 COMMAND 基类

```js
// os/command.js:7-30
COMMAND = function () {
    this.allow_fight = true;      // ★ 默认允许战斗中执行
    this.allow_level = 0;
}
COMMAND.inherits(BASE);
COMMAND.prototype.create = function (fname) {
    this.command.split(',').forEach(k => {                 // ★ 逗号分隔 = 多别名
        if (WORLD.COMMANDS[k]) console.error("command %s 重复", fname);
        WORLD.COMMANDS[k] = this;                          // ⚠️ 重名仍会覆盖
    });
}
```

**基类没有 `enter` 方法** —— 完全由脚本自己定义。

别名实例：`this.command = "look,l"`、`this.command = "cha,skills"`、`this.command = "select,command"`。

## 13.2 指令脚本范式

```js
this.inherits(COMMAND);
this.command = "go";                    // 必填，可逗号分隔多个
this.regex = /(\w+)\s+(\w+)/;           // 可选，参数正则
this.allow_busy = true;                 // 可选六道闸门
this.enter = function (me, ...pars) { } // 业务入口
```

## 13.3 分发完整链路（★）

```
TCP data (os/net-ws.js:52)
  ↓ protocol.readData → server.onReceive(msg, socket)
  ↓ os/ws.js:23 → WORLD.request(msg, socket)          os/world.js:75
      ├─ request_count > 20 → "不要急，慢慢来。"
      └─ user.command(request)
  ↓ os/char/character.js:136 CHARACTER.command(req)
      ├─ this.wait_input 存在 → wait_input(this, req) 直接返回（登录期/等待输入态）
      └─ 按第一个空格切出 cmd / pars → do_command(cmd, pars)
  ↓ os/char/character.js:156 do_command(cmdName, str)
      ├─ cmd = WORLD.COMMANDS[cmdName]
      ├─ cmd.regex && str → pars = cmd.regex.exec(str); pars[0] = this
      │    （★ 注意：把 pars[0] 从"完整匹配串"改成了 me）
      │    否则 pars = [this, str]
      ├─ ① do_item_action(this.environment, cmdName, pars)   ← 房间优先
      ├─ ② cmd = cmd || WORLD.DEFAULT_COMMAND                 ⚠️ 该变量从未赋值
      │    check_command(cmd)  六道闸门
      │    cmd.enter.apply(cmd, pars) !== false → 结束
      ├─ ③ str 存在 → do_item_action(this.find_obj(str, environment), cmdName, pars)
      └─ ④ 都不匹配 → this.send("什么？")
```

### 三级查找顺序的战略意义

1. **房间/环境动作优先**：房间可以用 `add_action("dazuo", ...)` 覆盖同名全局指令，实现"在这个房间里打坐有事发生"
2. **全局指令**
3. **目标物件动作兜底**：`kill xiong` 中 `xiong` 是目标，若全局无 `kill xiong` 则查目标身上的 action

`do_item_action(item, cmd, pars)`（`character.js:189`）：命中 `item.actions[cmd].action` 时 `apply(item, pars)`，返回 `true` 即中断分发。

## 13.4 六道权限/状态闸门（character.js:198-222）

```js
CHARACTER.prototype.check_command = function (cmd) {
    if (cmd.allow_level   > this.user_level) { this.send("什么？"); return false; }
    if (this.hp <= 0 && !cmd.allow_die)      return this.notify_fail("你现在是灵魂状态，不能那么做。");
    if (!cmd.allow_faint && this.is_faint)   { this.send("你正在昏迷中！"); return false; }
    if (!cmd.allow_state && this.state)      return this.notify_fail("你正在" + this.state.title + "，没时间这么做。");
    if (!cmd.allow_fight && this.fight_type > 0) return this.notify_fail("你正在战斗，待会再说。");
    if (!cmd.allow_busy  && this.is_busy)    return this.notify_fail("你现在正忙。");
    return true;
}
```

| 开关 | 默认值 | 含义 | 使用文件数 |
|---|---|---|---:|
| `allow_level` | 0 | 最低账号等级（GM 等级） | 8（值分布 9/6×3/5/3/2/1） |
| `allow_busy` | false | 忙乱中可用 | 65 |
| `allow_state` | false | 打坐/疗伤中可用 | 65 |
| `allow_die` | false | 灵魂状态可用 | 53 |
| `allow_faint` | false | 昏迷中可用 | 22 |
| `allow_fight` | **true** | 战斗中可用 | 15 |
| `allow_login` | false | **登录未完成阶段可执行**（供 `login`/`createrole` 用） | — |

> ❌ `this.admin = true`（5 个文件使用：`cmd/admin/recover.js`、`cmd/extend/{auto,ex,paimai,party}.js`）**全项目无任何消费点**，权限实际只由 `allow_level` 控制。

## 13.5 参数解析

### 方式一：空格切分（默认）

`command()` 找第一个非空字符段作为 cmd，其余作为 `pars` 字符串。

### 方式二：`this.regex`（58/123 指令使用）

```js
// world/cmd/dialog/task.js
this.inherits(COMMAND);
this.command = "task";
this.regex = /(\w+)\s+(\w+)(?:\s(\w+))?(?:\s(\w+))?/;
this.enter = function (me, tid, cmd, oid) {    // ★ 4 个捕获组直接映射到形参
    if (tid === 'all') return this.tasks_fin(me);
    let task = USERTASK.GET(tid);
}
```

> 因为 `do_command` 里做了 `pars[0] = this`，所以 `enter` 的第 1 参永远是 `me`，第 2..N 参是捕获组。

## 13.7 三类真实示例

### admin — GM 热更新指令（world/cmd/admin/update.js）

```js
this.inherits(COMMAND);
this.command = "update";
this.allow_busy = true; this.allow_state = true; this.allow_die = true;
this.allow_level = 6;                              // ★ GM 权限
this.enter = function (me, arg) {
    if (!arg) return this.notify(me, "路径不合理");
    if (arg === 'room') return this.update_room(me);
    if (arg === 'npc')  return this.update_npc(me);
    arg = arg.replace(/\\/g, "/");
    var index = arg.indexOf("/");
    var path  = __PATH.WORLD + arg.substr(0, index + 1);   // "./world/cmd/"
    var fname = arg.substr(index + 1);                      // "admin/test"
    BASE.UPDATE(path, fname);
    this.notify(me, arg + "已更新。");
}
```

### comm — 信息查询（world/cmd/comm/info.js）

```js
this.inherits(COMMAND);
this.command = "info";
this.allow_busy = true; this.allow_state = true; this.allow_die = true;
this.enter = function (me, tid) {
    let notifier = me;
    if (tid && me.user_level > 0) {                // ★ 指令内二次权限判断，可代查他人
        me = WORLD.getUser(tid);
        if (!me) return notifier.send('没有这个角色。');
    }
    let str = [];
    str.push(me.color_name, '\n');
    str.push('武道塔进度', me.query_temp('wd_level', 0), '层...');
    notifier.send(str.join(""));
}
```

### battle — 击杀（world/cmd/battle/kill.js，节选）

```js
this.inherits(COMMAND);
this.command = "kill";
this.enter = function (me, argid) {
    if (me.state) return me.notify("你现在正在" + me.state.title + ",没时间杀人。");
    if (me.environment.no_fight) return me.notify("这里不允许战斗。");
    var target = me.find_obj(argid, me.environment);
    if (!target || !target.do_kill || target === me) return me.notify("你要攻击谁？");
    if (target.is_player) {
        if (target.query_setting("ban_pk")) return me.notify(target.name + "正在开启PK保护...");
        if (me.query_setting("ban_pk"))     return me.notify("你正在开启PK保护，无法主动PK玩家。");
        if (!me.query_temp("pk")) me.notify("<hic>你主动击杀玩家，7天内不能开启PK保护。</hic>");
        me.set_temp("pk", 1, 24 * 3600000 * 7);
    }
    me.send_room("\n$N对著$n喝道：「...今日不是你死就是我活！」\n", target);
    if (target.on_kill && target.on_kill(me) == false) return;
    me.do_kill(target);
    WORLD.auto_pfm(me, target);
}
```

## 13.9 等待输入态（`wait_input`）

`CHARACTER.wait_input` 是一个"下一个输入由谁处理"的钩子，用于**多步交互**（登录、确认框）：

```js
// 登录期
socket.user.wait_input = USERLOGIN.check_session.bind(USERLOGIN);   // os/world.js:53
// 拉完角色列表后
user.wait_input = USERLOGIN.wait_login;                              // extends/login.js:56
// 进入游戏后
user.wait_input = null;
```

`wait_login` 只允许 `allow_login` 的指令通过（`extends/login.js:59-71`），即登录完成前只能执行 `login` / `createrole` / `deleterole`。

## 13.10 复刻建议

1. ✅ **逗号别名**值得抄，一行搞定 `look,l`、`i,inv,inventory`。
2. ✅ **`allow_*` 六道闸门集中校验**值得抄：每个指令声明自己的可用条件，避免每个指令里写重复的 `if (me.is_busy) return`。
3. ✅ **`regex` 可选参数解析**值得抄：简单指令用空格切分，复杂指令用正则捕获组直接映射形参。
4. ✅ **三级查找（房间 → 全局 → 目标物件）**是 MUD 的经典设计，让"场景化指令"成为可能，值得抄。
5. ❌ `WORLD.DEFAULT_COMMAND` 全项目从未赋值（`character.js:170`），"未知命令"兜底永远是 `undefined` → 要么实现要么删掉。
6. ❌ 指令重名只 `console.error` 仍覆盖（`command.js:25`）→ 改为启动时**直接抛错终止**，或保留两者并报警。
7. ❌ `this.admin` 死标记 → 删除或实现。
8. ⚠️ 建议新增：指令自动注册到 GM 后台的指令列表（当前 `os/world.js` 的 `getCommandList` 读 `cmd.allow_level/desc/help`，与实际字段对不上）。
